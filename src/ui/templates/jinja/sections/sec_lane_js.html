{#- The shared detail lane: how a page shows and hides its column-3 panes, plus the guarded binder. Split out of sec_lcp_shared_js.html because the Capability matrix tab keeps prod's own editor script, which defines its OWN API and STATE — including both would be two `const API` declarations and a SyntaxError that kills the page. A page includes this ONE piece if all it needs is the lane. -#} // ── the shared detail lane ───────────────────────────────────────────────── // Used by Routing definitions → Arena boards and → Capability matrix. const PANE_TITLES = {}; const MP_DESKTOP = window.matchMedia('(min-width: 1100px)'); function mpPanel() { return document.getElementById('modelsPanel'); } // Show one pane (or none) and keep the lane's own state in step. // // DO NOT call mpSyncViewport() from here. mpSyncViewport() calls mpShow() on desktop, so the // pair recursed — mpShow → mpSyncViewport → mpShow → … — until `RangeError: Maximum call // stack size exceeded`. The symptom was silent and thoroughly misleading: every pane button // read as dead, and ✕ left the pane it was meant to clear, because each nested call // re-applied the OLD pane on its way back out. The pre-rework code never made this call; it // was added when this function was generalised, and the `mq = mq || MP_DESKTOP` guard further // down was a patch for the symptom rather than the cause. function mpSetPane(pane) { const panel = mpPanel(); if (!panel) return; pane = pane || ''; // Every pane carries `dp-pane` + `data-pane=`, so this hides them ALL before // revealing one. Two tabs have different pane sets, and a hardcoded pair of names cannot // serve both — which is what broke when the taxonomy pane moved to the matrix tab. panel.querySelectorAll('.dp-pane').forEach(el => { el.hidden = el.dataset.pane !== pane; }); // The lane's own empty state: the operator's rule is that ✕ leaves column 3 saying // NOTHING, rather than holding the rows that were open when it was pressed. const empty = document.getElementById('mpEmpty'); if (empty) empty.hidden = !!pane; const t = document.getElementById('mpTitle'); if (t) t.textContent = pane ? (PANE_TITLES[pane] || pane) : ''; panel.dataset.pane = pane; panel.classList.add('open'); // The backdrop is a MOBILE affordance: below 1100px the lane overlays the page and needs a // tap-target to dismiss. On desktop the lane is a reserved column, and a full-screen // backdrop there swallows clicks meant for column 2 — the bug reported on 2026-11-07 // ("clicking empty space made column 3 disappear"). Scoped to !desktop, as it was before. const back = document.getElementById('mpBackdrop'); if (back) back.classList.toggle('open', !MP_DESKTOP.matches); } function mpShow(pane) { mpSetPane(pane); } // Clear the CONTENT without deciding the lane's width. On desktop the lane is the reserved // column the operator asked for, so clearing is what ✕ means; column 2 must not reflow into // the freed space just because the panel was emptied. function mpClear() { mpSetPane(null); } function mpHide() { const panel = mpPanel(); if (!panel) return; panel.classList.remove('open'); const back = document.getElementById('mpBackdrop'); if (back) back.classList.remove('open'); } function mpClose() { // ✕ = "clear this panel". Desktop keeps the reserved lane (emptied, showing the hint); // below 1100px the lane is a slide-over, so ✕ also closes it. mpClear(); if (!MP_DESKTOP.matches) mpHide(); } // ── drag handle (file-explorer splitter), the shared lane mechanics ──────── // The dragged width rides --dp-width on ; --detail-width stays the shared // default owned by the overview block. (function mpResizer() { const handle = document.getElementById('mpResizer'); if (!handle) return; const MIN_W = 280, MAX_W = 760; let dragging = false; function setWidth(px, persist) { px = Math.max(MIN_W, Math.min(px, Math.min(window.innerWidth * 0.6, MAX_W))); document.body.style.setProperty('--dp-width', px + 'px'); if (persist) { try { localStorage.setItem('models-panel-width', String(px)); } catch (e) { /* private mode */ } } } try { const saved = parseInt(localStorage.getItem('models-panel-width') || '', 10); if (saved) setWidth(saved, false); } catch (e) { /* private mode */ } handle.addEventListener('pointerdown', ev => { dragging = true; handle.classList.add('dragging'); handle.setPointerCapture(ev.pointerId); document.body.classList.add('dp-resizing'); }); handle.addEventListener('pointermove', ev => { if (dragging) setWidth(window.innerWidth - ev.clientX, false); }); handle.addEventListener('pointerup', ev => { if (!dragging) return; dragging = false; handle.classList.remove('dragging'); handle.releasePointerCapture(ev.pointerId); document.body.classList.remove('dp-resizing'); setWidth(window.innerWidth - ev.clientX, true); }); handle.addEventListener('dblclick', () => { document.body.style.removeProperty('--dp-width'); try { localStorage.removeItem('models-panel-width'); } catch (e) { /* private mode */ } }); })(); function on(id, ev, fn) { const el = document.getElementById(id); if (el) el.addEventListener(ev, fn); } // Re-apply the lane's state for the viewport. Called by the matchMedia listener, which passes // the event — hence `mq` rather than a required query. ONE DIRECTION ONLY: this may call // mpShow/mpHide, and neither of those may call it back. function mpSyncViewport(mq) { const query = (mq && typeof mq.matches === 'boolean') ? mq : MP_DESKTOP; const panel = mpPanel(); if (!panel) return; if (query.matches) { // Desktop: the lane is the reserved column, so it stays — showing whichever pane is // chosen, or the empty state if the operator cleared it. mpSetPane(panel.dataset.pane || null); } else { mpHide(); // mobile: a fresh load shows the content, not a sheet over it } }
