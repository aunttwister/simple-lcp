A hardcoded pane list (pollPane/taxPane) cannot serve // two tabs whose panes differ — which is exactly what broke when the taxonomy pane moved // to the Capability matrix tab and left this function setting `hidden` on nothing. panel.querySelectorAll('.dp-pane').forEach(el => { el.hidden = el.dataset.pane !== pane; }); mpSyncViewport(); const t = document.getElementById('mpTitle'); if (t) t.textContent = PANE_TITLES[pane] || pane; panel.dataset.pane = pane; panel.classList.add('open'); const bd = document.getElementById('mpBackdrop'); if (bd) bd.classList.add('open'); } function mpHide() { const panel = mpPanel(); if (!panel) return; panel.classList.remove('open'); const back = document.getElementById('mpBackdrop'); if (back) back.classList.remove('open'); } function mpClose() { // ✕ CLOSES the lane on both breakpoints. It used to call mpShow('tax') on // desktop — a no-op whenever taxonomy was already the open pane, which is the // state this page now loads in, so the button read as broken. Closing is the // honest meaning for a control shaped like a close button; the two buttons in // column 2 bring the lane back, and a closed lane releases its reserved width // (see the #modelsPanel rule in dashboard.css) so nothing is wasted. mpHide(); } // ── drag handle (file-explorer splitter), the shared lane mechanics ──────── // The dragged width rides --dp-width on ; --detail-width stays the shared // default owned by the overview block. (function mpResizer() { const handle = document.getElementById('mpResizer'); if (!handle) return; const MIN_W = 280, MAX_W = 760; let dragging = false; function setWidth(px, persist) { px = Math.max(MIN_W, Math.min(px, Math.min(window.innerWidth * 0.6, MAX_W))); document.body.style.setProperty('--dp-width', px + 'px'); if (persist) { try { localStorage.setItem('models-panel-width', String(px)); } catch (e) { /* private mode */ } } } try { const saved = parseInt(localStorage.getItem('models-panel-width') || '', 10); if (saved) setWidth(saved, false); } catch (e) { /* private mode */ } handle.addEventListener('pointerdown', ev => { dragging = true; handle.classList.add('dragging'); handle.setPointerCapture(ev.pointerId); document.body.classList.add('dp-resizing'); }); handle.addEventListener('pointermove', ev => { if (dragging) setWidth(window.innerWidth - ev.clientX, false); }); handle.addEventListener('pointerup', ev => { if (!dragging) return; dragging = false; handle.classList.remove('dragging'); handle.releasePointerCapture(ev.pointerId); document.body.classList.remove('dp-resizing'); setWidth(window.innerWidth - ev.clientX, true); }); handle.addEventListener('dblclick', () => { document.body.style.removeProperty('--dp-width'); try { localStorage.removeItem('models-panel-width'); } catch (e) { /* private mode */ } }); })(); function on(id, ev, fn) { const el = document.getElementById(id); if (el) el.addEventListener(ev, fn); } function mpSyncViewport(mq) { // The media query is a DEFAULT, not a required argument: mpShow/mpHide/mpClose call this // with none, and `mq.matches` on undefined threw — which aborted mpShow BEFORE it set the // title or added .open, so the lane opened to an empty, closed panel. mq = mq || MP_DESKTOP; const panel = mpPanel(); if (!panel) return; // Growing to desktop must not force the lane open over an operator who closed // it; it restores the lane only if the pane was already chosen. if (mq.matches && panel.dataset.pane) mpShow(panel.dataset.pane);
